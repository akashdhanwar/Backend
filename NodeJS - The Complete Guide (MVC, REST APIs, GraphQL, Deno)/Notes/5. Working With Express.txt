(Lec-61) - Adding Middleware
    It is all about Middleware
    Request > Middleware(req,res,next)=>{} > next() > Middleware(req,res,next)=>{} > res.send() > Response

    const express = require("express");

    const app = express();

    // Add a Middleware, next is a function
    app.use((req, res, next) => {
      console.log("In Middleware");
      next();
    });
    // Will not be executed until next() from previous Middleware
    app.use((req, res, next) => {
      console.log("In Another Middleware");
    });

    app.listen(3000);

(Lec-62) - How Middleware Works
    // Will not be executed until next() from previous Middleware
    app.use((req, res, next) => {
      console.log("In Another Middleware");
      //   We need not to write headers, attached a body of type any
      res.send("<h1>Hello from Express!!</h1>");
    });
